local function LoadConfig()
    pcall(function()
        if isfile and readfile and isfile("FemboyHub_Config.json") then
            local str = readfile("FemboyHub_Config.json")
            local conf = HttpService:JSONDecode(str)
            if not conf then return end
            if conf.TargetPlayers ~= nil then _G.G_SilentAimTargetPlayers = conf.TargetPlayers end
            if conf.TargetMobs ~= nil then _G.G_SilentAimTargetMobs = conf.TargetMobs end
            if conf.TeamCheck ~= nil then _G.G_SilentAimTeamCheck = conf.TeamCheck end
            if conf.SkillAimbot ~= nil then _G.G_SilentAimSkill = conf.SkillAimbot end
            if conf.SoruAimbot ~= nil then _G.G_SoruAimbotEnabled = conf.SoruAimbot end
            if conf.SoruInf ~= nil then _G.G_SoruInfinitoEnabled = conf.SoruInf end
            if conf.Macro ~= nil then MacroEnabled = conf.Macro end
            if conf.MacroDelay ~= nil then MacroDelay = conf.MacroDelay / 1000 end
            
            if conf.AHK_Melee_Z ~= nil then AHK_Melee_Z = conf.AHK_Melee_Z end
            if conf.AHK_Melee_X ~= nil then AHK_Melee_X = conf.AHK_Melee_X end
            if conf.AHK_Melee_C ~= nil then AHK_Melee_C = conf.AHK_Melee_C end
            if conf.AHK_Fruit_Z ~= nil then AHK_Fruit_Z = conf.AHK_Fruit_Z end

            if conf.AHK_Fruit_C ~= nil then AHK_Fruit_C = conf.AHK_Fruit_C end

function addToggleElement(parent, labelText, defaultState, yPos, callback)
    local frame = Instance.new("Frame", parent)
    frame.Size = UDim2.new(1, -12, 0, 20)
    frame.Position = UDim2.new(0, 6, 0, yPos)
    frame.BackgroundTransparency = 1
    local label = Instance.new("TextLabel", frame)
    label.Size = UDim2.new(0.7, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = labelText
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 9

    local clickBtn = Instance.new("TextButton", frame)
    clickBtn.Size = UDim2.new(0, 36, 0, 16)
    clickBtn.Position = UDim2.new(1, -38, 0.5, -8)
    clickBtn.BackgroundColor3 = defaultState and currentThemeColor or Color3.fromRGB(25, 25, 30)
    clickBtn.Text = defaultState and "ON" or "OFF"
    clickBtn.Font = Enum.Font.GothamBold
    clickBtn.TextSize = 8
    clickBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    Instance.new("UICorner", clickBtn).CornerRadius = UDim.new(0, 6)
    Instance.new("UIStroke", clickBtn).Color = currentThemeColor

    local state = defaultState
    clickBtn.MouseButton1Click:Connect(function()
        state = not state
        clickBtn.BackgroundColor3 = state and currentThemeColor or Color3.fromRGB(25, 25, 30)
        clickBtn.Text = state and "ON" or "OFF"
        callback(state)
    end)
    return clickBtn
end

            for _,n in ipairs(FLASH_NAMES) do
                if string.find(string.lower(track.Name), string.lower(n)) then return true end
            end
            local id = track.Animation and track.Animation.AnimationId:match("%d+") or ""

            end

        end

                if not VirtualInputManager then return end
                VirtualInputManager:SendKeyEvent(true, kc, false, game)
                task.wait(0.001)
                VirtualInputManager:SendKeyEvent(false, kc, false, game)
            end

        function monitorCharMacro(char)
            local h = char:WaitForChild("Humanoid", 5)

        local currentSilentAimTarget = nil
        
        local function GetClosestTargetToCenter()
            local myHRP = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not myHRP then return nil end
            local maxDist3D = maxRange
            local closestPart, shortest3DDist = nil, maxDist3D
            
            local function checkTargetPart(character)
                if not character or not character:FindFirstChild("HumanoidRootPart") then return end
                local p = Players:GetPlayerFromCharacter(character)
                if p and p == player then return end
                if _G.G_SilentAimTeamCheck and p and p.Team and p.Team.Name == player.Team.Name then return end
                local hum = character:FindFirstChildOfClass("Humanoid")
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if hum and hum.Health > 0 and hrp then
                    local worldDist = (hrp.Position - myHRP.Position).Magnitude
                    if worldDist <= shortest3DDist then
                        shortest3DDist = worldDist; closestPart = hrp
                    end
                end
            end
            if _G.G_SilentAimTargetPlayers then for _, p in ipairs(Players:GetPlayers()) do if p ~= player then checkTargetPart(p.Character) end end end
            if _G.G_SilentAimTargetMobs then
                local enemies = workspace:FindFirstChild("Enemies")
                if enemies then for _, enemy in ipairs(enemies:GetChildren()) do checkTargetPart(enemy) end end
            end
            return closestPart
        end

                            if key == "Target" then return currentSilentAimTarget end
                        end
                    end
                    return oldIndex(self, key)
                end))

                    local method = ncm and tostring(ncm):lower() or ""

                            if typeof(arg) == "CFrame" then args[i] = CFrame.new(activePos) end
                        end
                        return oldNamecall(self, unpack(args))
                    end
                    return oldNamecall(self, ...)
                end))
            end)
        end

        end
    end)
